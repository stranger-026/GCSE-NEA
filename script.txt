function getCards() {
  return fetch("http://localhost:5000/card_info_request")
    .then(function (response) {
      if (!response.ok) {
        throw new Error("Network response was not OK");
      }
      return response.json();
    })
    .catch(function (error) {
      // Handle any errors
      console.error("Error:", error);
    });
}
